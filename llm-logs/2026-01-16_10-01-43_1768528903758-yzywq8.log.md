```sql
        ON PL.PURCHID   = PT.PURCHID
       AND PL.DATAAREAID = PT.DATAAREAID
    WHERE PT.CREATEDDATETIME >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)
      AND PT.CREATEDDATETIME <  DATEFROMPARTS(YEAR(GETDATE()) + 1, 1, 1)
      -- AND PT.DATAAREAID = 'USMF'  -- uncomment / adjust to limit to one company
),
PO_Totals AS (
    SELECT
        DATAAREAID,
        VendorAccountNum,
        VendorName,
        SUM(POLineAmountMST) AS TotalPOAmount
    FROM ThisYearPOs
    GROUP BY DATAAREAID, VendorAccountNum, VendorName
),
Receipt_Totals AS (
    SELECT
        P.DATAAREAID,
        P.VendorAccountNum,
        SUM(PST.LINEAMOUNTMST) AS TotalReceiptAmount
    FROM ThisYearPOs P
    JOIN VENDPACKINGSLIPTRANS PST
        ON PST.PURCHID    = P.PURCHID
       AND PST.LINENUM    = P.LINENUM
       AND PST.DATAAREAID = P.DATAAREAID
    WHERE PST.PACKINGSLIPDATE >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)
      AND PST.PACKINGSLIPDATE <  DATEFROMPARTS(YEAR(GETDATE()) + 1, 1, 1)
    GROUP BY P.DATAAREAID, P.VendorAccountNum
),
Invoice_Totals AS (
    SELECT
        P.DATAAREAID,
        P.VendorAccountNum,
        SUM(VIT.LINEAMOUNTMST) AS TotalInvoiceAmount
    FROM ThisYearPOs P
    JOIN VENDINVOICETRANS VIT
        ON VIT.PURCHID    = P.PURCHID
       AND VIT.LINENUM    = P.LINENUM
       AND VIT.DATAAREAID = P.DATAAREAID
    WHERE VIT.INVOICEDATE >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)
      AND VIT.INVOICEDATE <  DATEFROMPARTS(YEAR(GETDATE()) + 1, 1, 1)
    GROUP BY P.DATAAREAID, P.VendorAccountNum
),
Payment_Totals AS (
    -- All vendor payments this year (not strictly tied to specific POs)
    SELECT
        VT.DATAAREAID,
        VT.ACCOUNTNUM AS VendorAccountNum,
        SUM(CASE WHEN VT.AMOUNTMST < 0 THEN -VT.AMOUNTMST ELSE 0 END) AS TotalPaymentAmount
    FROM VENDTRANS VT
    WHERE VT.TRANSDATE >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)
      AND VT.TRANSDATE <  DATEFROMPARTS(YEAR(GETDATE()) + 1, 1, 1)
      -- Optionally restrict to payment transaction types only, e.g.: 
      -- AND VT.TRANSTYPE = 2  -- example: 2 = Payment (confirm in your enum)
    GROUP BY VT.DATAAREAID, VT.ACCOUNTNUM
),
PR_Totals AS (
    -- Purchase requisition amounts traced back from PO lines
    SELECT
        P.DATAAREAID,
        P.VendorAccountNum,
        SUM(PRL.LINEAMOUNTMST) AS TotalPRAmount
    FROM ThisYearPOs P
    JOIN PURCHREQLINE PRL
        ON PRL.PURCHREQID      = P.PURCHREQID
       AND PRL.LINENUM         = P.PURCHREQLINENUM
       AND PRL.DATAAREAID      = P.DATAAREAID
    JOIN PURCHREQTABLE PRT
        ON PRT.PURCHREQID      = PRL.PURCHREQID
       AND PRT.DATAAREAID      = PRL.DATAAREAID
    WHERE PRT.CREATEDDATETIME >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)
      AND PRT.CREATEDDATETIME <  DATEFROMPARTS(YEAR(GETDATE()) + 1, 1, 1)
    GROUP BY P.DATAAREAID, P.VendorAccountNum
)
SELECT TOP 10
    PO.DATAAREAID,
    PO.VendorAccountNum        AS [Vendor Account Number],
    PO.VendorName              AS [Vendor Name],
    PO.TotalPOAmount           AS [Total PO Amount],
    ISNULL(RT.TotalReceiptAmount, 0) AS [Receipt Amount],
    ISNULL(IT.TotalInvoiceAmount, 0) AS [Invoiced Amount],
    ISNULL(PYT.TotalPaymentAmount, 0) AS [Payments Amount],
    ISNULL(PRT.TotalPRAmount, 0)      AS [PR Amount]
FROM PO_Totals PO
LEFT JOIN Receipt_Totals RT
    ON RT.DATAAREAID       = PO.DATAAREAID
   AND RT.VendorAccountNum = PO.VendorAccountNum
LEFT JOIN Invoice_Totals IT
    ON IT.DATAAREAID       = PO.DATAAREAID
   AND IT.VendorAccountNum = PO.VendorAccountNum
LEFT JOIN Payment_Totals PYT
    ON PYT.DATAAREAID      = PO.DATAAREAID
   AND PYT.VendorAccountNum = PO.VendorAccountNum
LEFT JOIN PR_Totals PRT
    ON PRT.DATAAREAID      = PO.DATAAREAID
   AND PRT.VendorAccountNum = PO.VendorAccountNum
ORDER BY PO.TotalPOAmount DESC;
```

### Explanation

The query below is tailored for a Dynamics AX / D365FO SQL database, using typical Ax table names (VendTable, PurchTable, PurchLine, etc.). It gets this-year POs, then aggregates PO, receipt, invoice, payment, and related purchase requisition amounts by vendor, returning the top 10 vendors by total PO amount.
