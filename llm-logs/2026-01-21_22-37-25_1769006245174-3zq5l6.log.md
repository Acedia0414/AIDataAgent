### Generated SQL

```sql
WITH TopCustomers AS (
    SELECT TOP 10
        ci.CustAccount
    FROM
        CustInvoiceJour ci
    WHERE
        ci.InvoiceDate BETWEEN '2025-01-01' AND '2025-03-31'
    GROUP BY
        ci.CustAccount
    ORDER BY
        SUM(ci.AmountCur) DESC
)
SELECT
    ct.AccountNum,
    ct.Name AS CustomerName,
    ci.InvoiceId,
    ci.InvoiceDate,
    ci.DueDate,
    ci.AmountCur AS OriginalAmount,
    ci.RemainAmountCur AS RemainingAmount,
    ci.CurrencyCode,
    DATEDIFF(day, ci.DueDate, GETDATE()) AS DaysPastDue,
    DATENAME(month, ci.InvoiceDate) AS InvoiceMonth,
    YEAR(ci.InvoiceDate) AS InvoiceYear
FROM
    CustInvoiceJour ci
JOIN
    CustTable ct ON ci.CustAccount = ct.AccountNum AND ci.DataAreaId = ct.DataAreaId
WHERE
    ci.CustAccount IN (SELECT CustAccount FROM TopCustomers)
    AND ci.DueDate < GETDATE() -- Overdue invoices
    AND ci.RemainAmountCur > 0 -- Unpaid portion
    AND ci.InvoiceDate BETWEEN '2025-01-01' AND '2025-03-31' -- Invoices issued in Q1 2025
ORDER BY
    ct.Name, ci.DueDate;
```
