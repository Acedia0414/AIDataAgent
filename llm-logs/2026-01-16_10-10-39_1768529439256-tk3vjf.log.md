WITH ThisYearPOs AS (
    SELECT
        PT.DATAAREAID,
        PT.PURCHID,
        PL.LINENUMBER                 AS POLineNumber,
        V.ACCOUNTNUM                  AS VendorAccountNum,
        PL.LINEAMOUNT                 AS POLineAmount,
        PT.CREATEDDATETIME            AS POCreateDateTime
    FROM PURCHTABLE PT
    JOIN VENDTABLE V
        ON V.ACCOUNTNUM = PT.ORDERACCOUNT
       AND V.DATAAREAID = PT.DATAAREAID
    JOIN PURCHLINE PL
        ON PL.PURCHID    = PT.PURCHID
       AND PL.DATAAREAID = PT.DATAAREAID
    WHERE PT.CREATEDDATETIME >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)
      AND PT.CREATEDDATETIME <  DATEFROMPARTS(YEAR(GETDATE()) + 1, 1, 1)
      -- AND PT.DATAAREAID = 'USMF'  -- optional: restrict to a single company
),
PO_Totals AS (
    SELECT
        DATAAREAID,
        VendorAccountNum,
        SUM(POLineAmount) AS TotalPOAmount
    FROM ThisYearPOs
    GROUP BY DATAAREAID, VendorAccountNum
),
Receipt_Totals AS (
    SELECT
        ty.DATAAREAID,
        ty.VendorAccountNum,
        SUM(vpst.VALUEMST) AS TotalReceiptAmount
    FROM ThisYearPOs ty
    JOIN VENDPACKINGSLIPTRANS vpst
        ON vpst.ORIGPURCHID            = ty.PURCHID
       AND vpst.PURCHASELINELINENUMBER = ty.POLineNumber
       AND vpst.DATAAREAID             = ty.DATAAREAID
    WHERE vpst.ACCOUNTINGDATE >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)
      AND vpst.ACCOUNTINGDATE <  DATEFROMPARTS(YEAR(GETDATE()) + 1, 1, 1)
    GROUP BY ty.DATAAREAID, ty.VendorAccountNum
),
Invoice_Totals AS (
    SELECT
        ty.DATAAREAID,
        ty.VendorAccountNum,
        SUM(vit.LINEAMOUNTMST) AS TotalInvoiceAmount
    FROM ThisYearPOs ty
    JOIN VENDINVOICETRANS vit
        ON vit.PURCHID                = ty.PURCHID
       AND vit.PURCHASELINELINENUMBER = ty.POLineNumber
       AND vit.DATAAREAID             = ty.DATAAREAID
    WHERE vit.INVOICEDATE >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)
      AND vit.INVOICEDATE <  DATEFROMPARTS(YEAR(GETDATE()) + 1, 1, 1)
    GROUP BY ty.DATAAREAID, ty.VendorAccountNum
),
Payment_Totals AS (
    -- All vendor cash outflows this year (negative AMOUNTMST summed as positive)
    SELECT
        vt.DATAAREAID,
        vt.ACCOUNTNUM AS VendorAccountNum,
        SUM(CASE WHEN vt.AMOUNTMST < 0 THEN -vt.AMOUNTMST ELSE 0 END) AS TotalPaymentAmount
    FROM VENDTRANS vt
    WHERE vt.TRANSDATE >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)
      AND vt.TRANSDATE <  DATEFROMPARTS(YEAR(GETDATE()) + 1, 1, 1)
    GROUP BY vt.DATAAREAID, vt.ACCOUNTNUM
)
SELECT TOP 10
    po.DATAAREAID,
    po.VendorAccountNum                 AS [Vendor Account Number],
    po.TotalPOAmount                    AS [Total PO Amount],
    ISNULL(rt.TotalReceiptAmount, 0)    AS [Receipt Amount],
    ISNULL(it.TotalInvoiceAmount, 0)    AS [Invoiced Amount],
    ISNULL(pyt.TotalPaymentAmount, 0)   AS [Payments Amount]
FROM PO_Totals po
LEFT JOIN Receipt_Totals rt
    ON rt.DATAAREAID       = po.DATAAREAID
   AND rt.VendorAccountNum = po.VendorAccountNum
LEFT JOIN Invoice_Totals it
    ON it.DATAAREAID       = po.DATAAREAID
   AND it.VendorAccountNum = po.VendorAccountNum
LEFT JOIN Payment_Totals pyt
    ON pyt.DATAAREAID      = po.DATAAREAID
   AND pyt.VendorAccountNum = po.VendorAccountNum
ORDER BY po.TotalPOAmount DESC;
