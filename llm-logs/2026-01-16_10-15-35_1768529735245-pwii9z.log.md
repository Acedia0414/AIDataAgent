),
PO_Totals AS (
    SELECT
        DATAAREAID,
        VendorAccountNum,
        SUM(POLineAmount) AS TotalPOAmount
    FROM ThisYearPOs
    GROUP BY DATAAREAID, VendorAccountNum
),
Receipt_Totals AS (
    SELECT
        ty.DATAAREAID,
        ty.VendorAccountNum,
        SUM(vpst.VALUEMST) AS TotalReceiptAmount

       AND vpst.PURCHASELINELINENUMBER = ty.POLineNumber
       AND vpst.DATAAREAID             = ty.DATAAREAID
    WHERE vpst.ACCOUNTINGDATE >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)

       AND vit.PURCHASELINELINENUMBER = ty.POLineNumber
       AND vit.DATAAREAID             = ty.DATAAREAID
