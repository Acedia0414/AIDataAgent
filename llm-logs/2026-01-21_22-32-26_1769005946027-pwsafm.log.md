### Generated SQL

SELECT TOP 50
    AT.AssetId,
    AT.Name AS AssetName,
    AT.AssetGroup,
    ABC.DepreciationEndDate,
    ABC.NetBookValue AS CurrentBookValue,
    ABC.AcquisitionPrice AS CurrentAcquisitionPrice,
    ABT.NetBookValue AS TaxBookValue,
    ABT.AcquisitionPrice AS TaxAcquisitionPrice,
    (ABC.NetBookValue - ABT.NetBookValue) AS NetBookValueDifference,
    CASE
        WHEN ABC.NetBookValue <= 0.001 AND ABC.AcquisitionPrice > 0 THEN 'Review for Impairment (Book Value Near Zero)'
        ELSE 'No Impairment Review Indicated'
    END AS ImpairmentReviewFlag
FROM
    AssetTable AT
JOIN
    AssetBook ABC ON AT.AssetId = ABC.AssetId
LEFT JOIN
    AssetBook ABT ON AT.AssetId = ABT.AssetId AND ABT.AssetBookId = 'TAX' -- Assumes 'TAX' is the tax book ID
WHERE
    ABC.AssetBookId = 'CURRENT' -- Assumes 'CURRENT' is the main financial book ID
    AND ABC.DepreciationEndDate >= DATEADD(month, DATEDIFF(month, 0, GETDATE()) + 1, 0) -- Start of next month
    AND ABC.DepreciationEndDate < DATEADD(month, DATEDIFF(month, 0, GETDATE()) + 2, 0) -- End of next month
ORDER BY
    ABC.DepreciationEndDate ASC;
